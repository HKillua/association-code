```bash
set -euo pipefail
task_repo=/home/wangguangjie/djs/association-code
task_python=/home/wangguangjie/djs/mtad-field-20261016/bin/python

task_out="$task_repo/runs/LEVEL1-RAW-CASE-CHAIN-20261027-105-attempt-1"
cd "$task_repo"
git fetch origin
test ! -e "$task_code"
test ! -e "$task_out"
test ! -e "$task_out-launch"
mapfile -t task_archives < <("$task_python" - "$task_original/execution.json" <<'PY'
import json,sys
from pathlib import Path
parts=json.load(open(sys.argv[1]))['arguments']['archive']
if not isinstance(parts,list) or not parts or not all(Path(p).is_file() for p in parts):
    raise SystemExit('Original Level1 archive paths missing; preserve and resolve before starting')
print('\n'.join(parts))
PY
)
test "${#task_archives[@]}" -gt 0
git worktree add --detach "$task_code" origin/main
"$task_python" "$task_code/tools/experiments/launch_detached.py" \
  --cwd "$task_code" --record-dir "$task_out-launch" \
  --time-policy advisory --max-seconds 900 --grace-seconds 30 \
  --env CUDA_VISIBLE_DEVICES= --env OMP_NUM_THREADS=4 --env MKL_NUM_THREADS=4 --env OPENBLAS_NUM_THREADS=4 \
  -- "$task_python" "$task_code/tools/analysis/audit_raw_case_chain.py" \
  --archive "${task_archives[@]}" --manifest "$task_code/configs/data/mtad-level1-scene3000.json" \
  --source103 "$task_source103" --source104 "$task_source104" --output "$task_out"
```
