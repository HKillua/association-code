```bash
run_compare() (
  set -euo pipefail
  task_kind=$1
  task_python=/home/wangguangjie/djs/mtad-field-20261016/bin/python
  task_archive=/home/wangguangjie/djs/data/mtad/数据集level2.zip
  task_prepared=runs/PREP-NATIVE-20261017-007-attempt-1
  case "$task_kind" in
    cdf) task_cap=10800; task_out=runs/CDF1-VALIDATION-20261017-011-attempt-1
      task_args=(tools/experiments/cdf1_validation.py
        --reference runs/FIRST-FIT-20261017-010-attempt-1
        --manifest "$task_prepared/manifest.json" --config configs/first-fit-v2.json) ;;
    afull) task_cap=10800; task_out=runs/AFULL-20261017-011-attempt-1
      task_args=(-m association.cli train-reference --archive "$task_archive"
        --prepared "$task_prepared" --config configs/first-fit-v2.json
        --role A_full --branch ce --seed 20) ;;
    tsa) task_cap=21600; task_out=runs/TSACTFER-20261017-011-attempt-1
      task_args=(tools/experiments/tsactfer_fit.py --archive "$task_archive"
        --manifest "$task_prepared/manifest.json" --protocol-config configs/first-fit-v2.json
        --tsa-config configs/tsactfer-first-fit.json) ;;
    *) return 2 ;;
  esac
  task_root="HandOver/results/COMPARE-20261017-011/$task_kind"
  test ! -e "$task_out"
  mkdir -p "$(dirname "$task_root")"
  mkdir "$task_root"
  task_started=$(date +%s)
  task_deadline=$(( task_started + task_cap ))
  printf '%s\n' "$task_deadline" > "$task_root/deadline_epoch.txt"
  date -u +%FT%TZ > "$task_root/started_at.txt"
  task_rc=0
  timeout 45s "$task_python" tools/setup/gpu_allocation.py --device "${2:-auto}" \
    --output "$task_root/gpu.json" > "$task_root/gpu_uuid.txt" 2> "$task_root/gpu_stderr.log" || task_rc=$?
  if (( task_rc == 0 )); then
    IFS= read -r task_gpu_uuid < "$task_root/gpu_uuid.txt"
    task_remaining=$(( task_deadline - $(date +%s) - 15 ))
    (( task_remaining > 0 ))
    task_cmd=(env CUDA_VISIBLE_DEVICES="$task_gpu_uuid" OMP_NUM_THREADS=4 MKL_NUM_THREADS=4
      "$task_python" "${task_args[@]}" --device cuda:0 --cpu-threads 4 --output "$task_out")
    printf '%q ' "${task_cmd[@]}" > "$task_root/command.txt"
    printf '\n' >> "$task_root/command.txt"
    timeout --signal=TERM --kill-after=15s "${task_remaining}s" "${task_cmd[@]}" \
      > "$task_root/stdout.log" 2> "$task_root/stderr.log" || task_rc=$?
  fi
  printf '%s\n' "$task_rc" > "$task_root/exit_code.txt"
  date -u +%FT%TZ > "$task_root/ended_at.txt"
  printf '%s\n' "$(( $(date +%s) - task_started ))" > "$task_root/elapsed_seconds.txt"
  return "$task_rc"
)
```

命令失败后不重新调用函数。输入/目录/选卡失败同样回传实际错误与退出码，未训练就写未训练。若使用现场包装器，保留实际命令与包装器文本；不能改变训练参数或重置deadline。收到timeout后确认本作业子进程终止，再释放并发槽。
