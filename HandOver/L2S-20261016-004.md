- 输出为 `HandOver/results/ENV-20261016-004/attempt-N/`；CUDA 位于其中 `cuda/`。旧输出不删除、不覆盖。

在仓库根目录使用 Bash，先创建结果根目录并定义以下包装器；如保存为现场脚本，应放仓库外并在回执记录来源。函数定义本身不启动任务。

```bash
mkdir HandOver/results/ENV-20261016-004

env004_budget_start() (
  set -euo pipefail
  case "$1" in cpu) task_seconds=900 ;; cuda) task_seconds=600 ;; *) exit 2 ;; esac
  task_deadline_file="HandOver/results/ENV-20261016-004/$1_deadline_epoch.txt"
  if test ! -e "$task_deadline_file"; then
    (set -C; printf '%s\n' "$(( $(date +%s) + task_seconds ))" > "$task_deadline_file")
  fi
)

env004_run() (
  set -euo pipefail
  task_phase=$1
  task_attempt=$2
  shift 2
  case "$task_attempt" in 1|2) ;; *) exit 2 ;; esac
  task_root=HandOver/results/ENV-20261016-004
  task_out="$task_root/attempt-$task_attempt"
  case "$task_phase" in
    cpu)
      test "$#" = 0
      task_script=tools/setup/verify_field_004.sh
      task_result="$task_out/setup_wrapper_exit_code.txt"
      set -- "$task_attempt"
      ;;
    cuda)
      test "$#" = 2
      task_gpu=$1
      task_cpu_attempt=$2
      task_script=tools/setup/cuda_smoke_004.sh
      task_result="$task_out/cuda_wrapper_exit_code.txt"
      set -- "$task_gpu" "$task_attempt" "$task_cpu_attempt"
      ;;
    *) exit 2 ;;
  esac
  task_deadline=$(cat "$task_root/${task_phase}_deadline_epoch.txt")
  [[ "$task_deadline" =~ ^[0-9]+$ ]]
  task_remaining=$(( task_deadline - $(date +%s) ))
  if (( task_remaining <= 30 )); then
    printf 'Insufficient time including termination grace\n' >&2
    exit 124
  fi
  # Never overwrite a previous invocation, even if the inner script failed early.
  (set -C; date -u +%FT%TZ > "$task_root/${task_phase}-attempt-${task_attempt}-started.txt")
  task_rc=0
  timeout --signal=TERM --kill-after=30s "$((task_remaining - 30))s" \
    bash "$task_script" "$@" \
    > "$task_root/${task_phase}-attempt-${task_attempt}-wrapper.log" 2>&1 || task_rc=$?
  printf '%s\n' "$task_rc" > "$task_root/${task_phase}-attempt-${task_attempt}-wrapper_exit_code.txt"
  if test -d "$task_out"; then
    printf '%s\n' "$task_rc" > "$task_result"
  fi
  exit "$task_rc"
)
```
